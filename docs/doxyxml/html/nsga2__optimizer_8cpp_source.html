126void NSGA2Optimizer::transformPopulation(size_t old_n_params, size_t new_n_params,
127 const std::vector<std::pair<size_t, size_t>>& parameter_mapping) {
128 /*
129 * Population transformation strategy inspired by:
130 * - "Multi-space evolutionary search" (Neural Comput. Appl. 2022)
131 * - "Dynamic multi-objective optimization for complex changes" (Knowl. Based Syst. 2021)
132 */
133
134 if (!population_ || population_->size() == 0) return;
135
136 // Bounds should already be updated at this point
137 if (config_.lower_bounds.size() != new_n_params || config_.upper_bounds.size() != new_n_params) {
138 std::cerr << "Error: Bounds size mismatch - lower_bounds.size()=" << config_.lower_bounds.size()
139 << ", upper_bounds.size()=" << config_.upper_bounds.size()
140 << ", new_n_params=" << new_n_params << std::endl;
141 throw std::runtime_error("Bounds not properly set before population transformation");
142 }
143
144 // Create new population with transformed individuals
145 auto new_population = std::make_shared<Population>();
146
147 // Validate parameter mappings
148 for (const auto& mapping : parameter_mapping) {
149 if (mapping.first >= old_n_params) {
150 std::cerr << "Error: Invalid old index " << mapping.first
151 << " for old parameter space size " << old_n_params << std::endl;
152 throw std::out_of_range("Invalid old parameter index in mapping");
153 }
154 if (mapping.second >= new_n_params) {
155 std::cerr << "Error: Invalid new index " << mapping.second
156 << " for new parameter space size " << new_n_params << std::endl;
157 throw std::out_of_range("Invalid new parameter index in mapping");
158 }
159 }
160
161 // Transform each individual
162 for (size_t i = 0; i < population_->size(); ++i) {
163 auto& old_ind = (*population_)[i];
164
165 // Create new individual with new parameter space size
166 auto new_ind = std::make_shared<Individual>(new_n_params);
167
168 // Initialize all parameters with random values within new bounds
169 std::uniform_real_distribution<> dist(0.0, 1.0);
170 for (size_t j = 0; j < new_n_params; ++j) {
171 // Use stratified initialization based on population position
172 double interval_pos = (i + dist(gen_)) / population_->size();
173 new_ind->parameters[j] = config_.lower_bounds[j] +
174 interval_pos * (config_.upper_bounds[j] - config_.lower_bounds[j]);
175 }
176
177 // Copy preserved parameters according to mapping
178 for (const auto& mapping : parameter_mapping) {
179 size_t old_idx = mapping.first;
180 size_t new_idx = mapping.second;
181
182 // Safety check for old individual parameter access
183 if (old_idx < old_ind->parameters.size()) {
184 new_ind->parameters[new_idx] = old_ind->parameters[old_idx];
185 } else {
186 std::cerr << "Warning: Old individual " << i << " has fewer parameters than expected" << std::endl;
187 }
188 }
189
190 // Clear objectives for re-evaluation
191 new_ind->objectives.clear();
192 new_ind->rank = 0;
193 new_ind->crowding_distance = 0.0;
194
195 new_population->add(new_ind);
196 }
197
198 // Replace old population
199 population_ = new_population;
200
201 if (config_.verbose) {
202 std::cout << "Population transformation complete. New population size: "
203 << new_population->size() << " with " << new_n_params << " parameters each" << std::endl;
204 }
205}

207// Traditional crowding distance implementation
208void TraditionalCrowdingDistance::calculate(std::vector<IndividualPtr>& front) {
209 size_t n = front.size();
210 if (n <= 2) {
211 for (auto& ind : front) {
212 ind->crowding_distance = std::numeric_limits<double>::infinity();
213 }
214 return;
215 }
216
217 // Initialize distances
218 for (auto& ind : front) {
219 ind->crowding_distance = 0.0;
220 }
221
222 size_t n_objectives = front[0]->objectives.size();
223
224 for (size_t obj = 0; obj < n_objectives; ++obj) {
225 // Sort by objective
226 #ifdef __cpp_lib_parallel_algorithm
227 std::sort(std::execution::par_unseq, front.begin(), front.end(),
228 [obj](const IndividualPtr& a, const IndividualPtr& b) {
229 return a->objectives[obj] < b->objectives[obj];
230 });
231 #else
232 std::sort(front.begin(), front.end(),
233 [obj](const IndividualPtr& a, const IndividualPtr& b) {
234 return a->objectives[obj] < b->objectives[obj];
235 });
236 #endif
237
238 // Boundary points
239 front[0]->crowding_distance = std::numeric_limits<double>::infinity();
240 front[n-1]->crowding_distance = std::numeric_limits<double>::infinity();
241
242 double min_obj = front[0]->objectives[obj];
243 double max_obj = front[n-1]->objectives[obj];
244 double range = max_obj - min_obj;
245
246 if (range == 0.0) continue;
247
248 for (size_t i = 1; i < n - 1; ++i) {
249 if (front[i]->crowding_distance != std::numeric_limits<double>::infinity()) {
250 double distance = (front[i+1]->objectives[obj] - front[i-1]->objectives[obj]) / range;
251 front[i]->crowding_distance += distance;
252 }
253 }
254 }
255}

261double EquidistantSelectionCrowdingDistance::calculateDistance(const Individual& a, const Individual& b) const {
262 double distance = 0.0;
263 for (size_t i = 0; i < a.objectives.size(); ++i) {
264 double diff = b.objectives[i] - a.objectives[i];
265 distance += diff * diff;
266 }
267 return std::sqrt(distance);

270void EquidistantSelectionCrowdingDistance::calculate(std::vector<IndividualPtr>& front) {
271 if (front.empty() || front.size() < 2) return;
272
273 size_t numSelect = (selection_size_ > 0) ? selection_size_ : front.size();
274
275 // Sort by lexicographic order of objectives
276 std::sort(front.begin(), front.end(), [](const IndividualPtr& a, const IndividualPtr& b) {
277 for (size_t i = 0; i < a->objectives.size(); ++i) {
278 if (std::abs(a->objectives[i] - b->objectives[i]) > std::numeric_limits<double>::epsilon()) {
279 return a->objectives[i] < b->objectives[i];
280 }
281 }
282 return false;
283 });
284
285 // Calculate path lengths
286 std::vector<double> D(front.size(), 0.0);
287
288 // Initialize crowding distances
289 for (auto& ind : front) {
290 ind->crowding_distance = 0.0;
291 }
292
293 // Calculate cumulative path length
294 for (size_t i = 1; i < front.size(); ++i) {
295 D[i] = D[i - 1] + calculateDistance(*front[i - 1], *front[i]);
296 }
297
298 double Dmax = D.back();
299 if (Dmax < std::numeric_limits<double>::epsilon()) {
300 std::cerr << "Warning: Total path length is near zero!" << std::endl;
301 return;
302 }
303
304 // Set boundary points
305 front[0]->crowding_distance = std::numeric_limits<double>::infinity();
306 front.back()->crowding_distance = std::numeric_limits<double>::infinity();
307
308 if (numSelect > 2) {
309 double equidistantStep = Dmax / (numSelect - 1);
310
311 for (size_t i = 1; i < numSelect - 1; ++i) {
312 double targetDistance = equidistantStep * i;
313
314 // Find closest point to target distance
315 size_t bestIdx = 1;
316 double minDiff = std::abs(D[1] - targetDistance);
317
318 for (size_t j = 2; j < front.size() - 1; ++j) {
319 double diff = std::abs(D[j] - targetDistance);
320 if (diff < minDiff) {
321 minDiff = diff;
322 bestIdx = j;
323 }
324 }
325
326 // Update crowding distance for selected point
327 if (front[bestIdx]->crowding_distance == 0.0) {
328 front[bestIdx]->crowding_distance = Dmax / numSelect;
329 }
330 }
331 }

346// FastNonDominatedSort implementation
347std::vector<std::vector<IndividualPtr>> FastNonDominatedSort::sort(Population& population) {
348 size_t n = population.size();
349 std::vector<std::set<size_t>> S(n);
350 std::vector<int> n_dominated(n, 0);
351 std::vector<std::vector<size_t>> front_indices;
352 std::vector<std::vector<IndividualPtr>> sorted_fronts;
353
354 // Reset all individuals
355 for (auto& ind : population) {
356 ind->reset();
357 }
358
359 // Calculate domination relationships
360 #pragma omp parallel for if(n > 100)
361 for (size_t i = 0; i < n; ++i) {
362 for (size_t j = 0; j < n; ++j) {
363 if (i == j) continue;
364
365 if (population[i]->dominates(*population[j])) {
366 #pragma omp critical
367 S[i].insert(j);
368 } else if (population[j]->dominates(*population[i])) {
369 #pragma omp atomic
370 n_dominated[i]++;
371 }
372 }
373 }
374
375 // Find first front
376 std::vector<size_t> current_front;
377 for (size_t i = 0; i < n; ++i) {
378 if (n_dominated[i] == 0) {
379 population[i]->rank = 0;
380 current_front.push_back(i);
381 }
382 }
383
384 front_indices.push_back(current_front);
385
386 // Find remaining fronts
387 int rank = 0;
388 while (!current_front.empty()) {
389 std::vector<size_t> next_front;
390
391 for (size_t i : current_front) {
392 for (size_t j : S[i]) {
393 n_dominated[j]--;
394 if (n_dominated[j] == 0) {
395 population[j]->rank = rank + 1;
396 next_front.push_back(j);
397 }
398 }
399 }
400
401 rank++;
402 current_front = next_front;
403 if (!current_front.empty()) {
404 front_indices.push_back(current_front);
405 }
406 }
407
408 // Convert indices to individual pointers
409 sorted_fronts.resize(front_indices.size());
410 for (size_t i = 0; i < front_indices.size(); ++i) {
411 for (size_t idx : front_indices[i]) {
412 sorted_fronts[i].push_back(population[idx]);
413 }
414 }
415
416 return sorted_fronts;
417}

422std::vector<std::vector<double>> LatinHypercubeSampler::sample(
423 size_t n_samples, size_t n_dims,
424 const std::vector<double>& lower_bounds,
425 const std::vector<double>& upper_bounds) {
426
427 std::vector<std::vector<double>> samples(n_samples, std::vector<double>(n_dims));
428
429 for (size_t dim = 0; dim < n_dims; ++dim) {
430 // Create intervals
431 std::vector<size_t> intervals(n_samples);
432 std::iota(intervals.begin(), intervals.end(), 0);
433 std::shuffle(intervals.begin(), intervals.end(), gen_);
434
435 // Sample within intervals
436 std::uniform_real_distribution<> dist(0.0, 1.0);
437 double range = upper_bounds[dim] - lower_bounds[dim];
438
439 for (size_t i = 0; i < n_samples; ++i) {
440 double interval_pos = (intervals[i] + dist(gen_)) / n_samples;
441 samples[i][dim] = lower_bounds[dim] + interval_pos * range;
442 }
443 }
444
445 return samples;
446}

454double REXCrossover::generateVShapedRandom(double a) {
455 std::uniform_real_distribution<> dist(0.0, 1.0);
456 double u = dist(gen_);
457
458 // V-shaped distribution: f(x) = |x|/a² for x in [-a, a]
459 // CDF: F(x) = 1/2 - x²/(2a²) for x < 0
460 // F(x) = 1/2 + x²/(2a²) for x ≥ 0
461 // Inverse transform:
462 if (u < 0.5) {
463 // Left side: x = -a*sqrt(1-2u)
464 return -a * std::sqrt(1.0 - 2.0 * u);
465 } else {
466 // Right side: x = a*sqrt(2u-1)
467 return a * std::sqrt(2.0 * u - 1.0);
468 }
469}

471std::vector<IndividualPtr> REXCrossover::crossover(
472 const std::vector<IndividualPtr>& parents,
473 size_t n_children) {
474
475 if (parents.size() < 2) {
476 throw std::invalid_argument("REX crossover requires at least 2 parents");
477 }
478
479 // Validate all parents have same parameter size
480 size_t n_dims = parents[0]->parameters.size();
481 for (size_t i = 1; i < parents.size(); ++i) {
482 if (parents[i]->parameters.size() != n_dims) {
483 throw std::invalid_argument("All parents must have the same number of parameters for REX crossover");
484 }
485 }
486 size_t n_parents = parents.size();
487 std::vector<IndividualPtr> children;
488
489 // Calculate center of mass (xg in the PDF)
490 std::vector<double> center(n_dims, 0.0);
491 for (const auto& parent : parents) {
492 for (size_t i = 0; i < n_dims; ++i) {
493 center[i] += parent->parameters[i];
494 }
495 }
496 for (double& c : center) {
497 c /= n_parents;
498 }
499
500 // Generate children according to REX(φ,n+k) from the PDF
501 for (size_t c = 0; c < n_children; ++c) {
502 auto child = std::make_shared<Individual>(n_dims);
503
504 // Initialize with center
505 for (size_t d = 0; d < n_dims; ++d) {
506 child->parameters[d] = center[d];
507 }
508
509 // Add deviations using specified distribution
510 if (dist_type_ == REXDistributionType::VShaped) {
511 // V-shaped distribution: a = sqrt(2/(n+k)) as stated in PDF
512 double a = std::sqrt(2.0 / n_parents);
513
514 for (size_t p = 0; p < n_parents; ++p) {
515 double xi = generateVShapedRandom(a);
516 for (size_t d = 0; d < n_dims; ++d) {
517 child->parameters[d] += xi * (parents[p]->parameters[d] - center[d]);
518 }
519 }
520 } else if (dist_type_ == REXDistributionType::Normal) {
521 // Normal distribution N(0, σ²) where σ² = 1/(n+k)
522 std::normal_distribution<> normal_dist(0.0, std::sqrt(1.0 / n_parents));
523
524 for (size_t p = 0; p < n_parents; ++p) {
525 double xi = normal_dist(gen_);
526 for (size_t d = 0; d < n_dims; ++d) {
527 child->parameters[d] += xi * (parents[p]->parameters[d] - center[d]);
528 }
529 }
530 } else { // Uniform distribution
531 // Uniform distribution U(-a, a) where a = sqrt(3/(n+k)) for variance = 1/(n+k)
532 double a = std::sqrt(3.0 / n_parents);
533 std::uniform_real_distribution<> uniform_dist(-a, a);
534
535 for (size_t p = 0; p < n_parents; ++p) {
536 double xi = uniform_dist(gen_);
537 for (size_t d = 0; d < n_dims; ++d) {
538 child->parameters[d] += xi * (parents[p]->parameters[d] - center[d]);
539 }
540 }
541 }
542
543 children.push_back(child);
544 }
545
546 return children;
547}

557 size_t n_select,

615void NSGA2Optimizer::environmental_selection(Population& combined_pop, Population& new_pop) {
616 auto sorted_fronts = sorting_algorithm_->sort(combined_pop);
617
618 new_pop.clear();
619
620 // Add fronts until population is filled
621 for (auto& front : sorted_fronts) {
622 if (new_pop.size() + front.size() <= config_.population_size) {
623 for (auto& ind : front) {
624 new_pop.add(ind);
625 }
626 } else {
627 // Calculate crowding distance for last front
628 crowding_calculator_->calculate(front);
629
630 // Sort by crowding distance
631 std::sort(front.begin(), front.end(),
632 [](const IndividualPtr& a, const IndividualPtr& b) {
633 return a->crowding_distance > b->crowding_distance;
634 });
635
636 // Add individuals until population is full
637 size_t remaining = config_.population_size - new_pop.size();
638 for (size_t i = 0; i < remaining; ++i) {
639 new_pop.add(front[i]);
640 }

642 }

646void NSGA2Optimizer::evaluate_batch(
647 std::vector<IndividualPtr>& individuals,
648 const std::vector<ObjectiveFunction>& objectives,
649 const std::function<std::vector<std::vector<double>>(const std::vector<std::vector<double>>&)>& batch_evaluator) {
650
651 if (batch_evaluator) {
652 // Batch evaluation
653 std::vector<std::vector<double>> param_batch;
654 for (const auto& ind : individuals) {
655 // Verify individual has correct parameter size
656 if (ind->parameters.size() != config_.lower_bounds.size()) {
657 std::cerr << "Error: Individual has " << ind->parameters.size()
658 << " parameters, expected " << config_.lower_bounds.size() << std::endl;
659 throw std::runtime_error("Individual parameter size mismatch");
660 }
661 param_batch.push_back(ind->parameters);
662 }
663
664 auto objectives_batch = batch_evaluator(param_batch);
665
666 // Verify returned objectives match individual count
667 if (objectives_batch.size() != individuals.size()) {
668 throw std::runtime_error("Batch evaluator returned wrong number of objectives");
669 }
670
671 for (size_t i = 0; i < individuals.size(); ++i) {
672 individuals[i]->objectives = objectives_batch[i];
673 }
674 } else {
675 // Individual evaluation (OpenMP parallelized)
676 #pragma omp parallel for schedule(dynamic)
677 for (size_t i = 0; i < individuals.size(); ++i) {
678 // Verify individual has correct parameter size
679 if (individuals[i]->parameters.size() != config_.lower_bounds.size()) {
680 #pragma omp critical
681 {
682 std::cerr << "Error: Individual " << i << " has "
683 << individuals[i]->parameters.size()
684 << " parameters, expected " << config_.lower_bounds.size() << std::endl;
685 }
686 continue; // Skip this individual
687 }
688
689 individuals[i]->objectives.clear();
690 for (const auto& obj_func : objectives) {
691 auto obj_values = obj_func(individuals[i]->parameters);
692 individuals[i]->objectives.insert(
693 individuals[i]->objectives.end(),
694 obj_values.begin(),
695 obj_values.end()
696 );
697 }
698 }
699 }

702std::pair<std::vector<std::vector<double>>, std::vector<std::vector<double>>>
703NSGA2Optimizer::optimize(const std::vector<ObjectiveFunction>& objectives,
704 std::function<void(size_t, const Population&)> callback,
705 const std::function<std::vector<std::vector<double>>(const std::vector<std::vector<double>>&)>& batch_evaluator)
706{
707 // Initialize
708 start_time_ = std::chrono::steady_clock::now();
709 best_fitness_history_.clear();
710 mean_fitness_history_.clear();
711 best_fitness_ = std::numeric_limits<double>::infinity();
712 current_generation_ = 0;
713
714 size_t n_params = config_.lower_bounds.size();
715
716 // Initialize population using LHS
717 auto initial_params = lhs_sampler_->sample(
718 config_.population_size, n_params,
719 config_.lower_bounds, config_.upper_bounds
720 );
721
722 population_ = std::make_shared<Population>();
723 for (const auto& params : initial_params)
724 {
725 auto ind = std::make_shared<Individual>(params);
726 population_->add(ind);
727 }
728
729 // Evaluate initial population
730 std::vector<IndividualPtr> initial_individuals;
731 for (auto& ind : *population_)
732 {
733 initial_individuals.push_back(ind);
734 }
735 evaluate_batch(initial_individuals, objectives, batch_evaluator);
736
737 // Main evolution loop
738 for (size_t gen = 0; gen <= config_.max_generations; ++gen)
739 {
740 current_generation_ = gen;
741
742
743 // Calculate statistics only if objectives are available
744 double min_fitness = std::numeric_limits<double>::max();
745 double sum_fitness = 0.0;
746 bool statistics_valid = true;
747
748 for (const auto& ind : *population_)
749 {
750 if (ind->objectives.empty())
751 {
752 std::cerr << "ERROR: Individual has empty objectives after evaluation" << std::endl;
753 statistics_valid = false;
754 break;
755 }
756
757 double fitness = 0.0;
758 std::vector<double> weights(objectives.size(), 1.0);
759
760 for (size_t i = 0; i < ind->objectives.size() && i < weights.size(); ++i)
761 {
762 fitness += weights[i] * ind->objectives[i];
763 }
764
765 min_fitness = std::min(min_fitness, fitness);
766 sum_fitness += fitness;
767 }
768
769 if (statistics_valid)
770 {
771 double mean_fitness = sum_fitness / population_->size();
772 best_fitness_ = std::min(best_fitness_, min_fitness);
773 best_fitness_history_.push_back(best_fitness_);
774 mean_fitness_history_.push_back(mean_fitness);
775
776 // Progress reporting
777 if (config_.verbose && gen % config_.progress_interval == 0)
778 {
779 auto elapsed = std::chrono::duration_cast<std::chrono::seconds>(
780 std::chrono::steady_clock::now() - start_time_
781 ).count();
782
783 std::cout << "\n--- NSGA-II Generation " << gen << " ---" << std::endl;
784 std::cout << "Best fitness: " << min_fitness << ", Mean fitness: " << mean_fitness << std::endl;
785 std::cout << "Elapsed time: " << elapsed << " seconds" << std::endl;
786 std::cout << "Crowding distance method: " << crowding_calculator_->getName() << std::endl;
787
788 if (gen >= config_.progress_interval && best_fitness_history_.size() >= config_.progress_interval)
789 {
790 double old_fitness = std::min(best_fitness_history_[gen - config_.progress_interval], std::numeric_limits<double>::max());
791 double improvement = (old_fitness - min_fitness) / old_fitness * 100.0;
792 std::cout << "Improvement (" << config_.progress_interval << " gen): " << improvement << "%" << std::endl;
793 }
794 }
795 } else
796 {
797 // If statistics are not valid, use dummy values
798 best_fitness_history_.push_back(best_fitness_);
799 mean_fitness_history_.push_back(std::numeric_limits<double>::max());
800 }
801
802
803 // Create offspring population
804 Population offspring_pop;
805 std::vector<IndividualPtr> all_children;
806
807 // IMPORTANT: Get current parameter size dynamically
808 size_t current_n_params = config_.lower_bounds.size();
809
810 while (offspring_pop.size() < config_.population_size)
811 {
812 // Select parents
813 auto parents = selection_strategy_->select(*population_, config_.n_parents, gen_);
814
815 // Verify all parents have correct parameter size
816 bool parents_valid = true;
817 for (const auto& parent : parents)
818 {
819 if (parent->parameters.size() != current_n_params)
820 {
821 std::cerr << "Warning: Parent has " << parent->parameters.size()
822 << " parameters, expected " << current_n_params << std::endl;
823 parents_valid = false;
824 break;
825 }
826 }
827
828 if (!parents_valid)
829 {
830 // Skip this parent set and try again
831 continue;
832 }
833
834 // REX crossover
835 std::vector<IndividualPtr> children;
836 try
837 {
838 children = rex_crossover_->crossover(parents, config_.n_children);
839 }
840 catch (const std::exception& e)
841 {
842 std::cerr << "Crossover error: " << e.what() << std::endl;
843 continue; // Try again with different parents
844 }
845
846 // Apply bounds with current parameter size
847 for (auto& child : children)
848 {
849 // Ensure child has correct size
850 if (child->parameters.size() != current_n_params)
851 {
852 std::cerr << "Warning: Child has " << child->parameters.size()
853 << " parameters, resizing to " << current_n_params << std::endl;
854 child->parameters.resize(current_n_params, 0.0);
855 }
856
857 // Apply bounds
858 for (size_t i = 0; i < current_n_params; ++i)
859 {
860 child->parameters[i] = std::max(config_.lower_bounds[i],
861 std::min(config_.upper_bounds[i], child->parameters[i]));
862 }
863 all_children.push_back(child);
864 offspring_pop.add(child);
865
866 if (offspring_pop.size() >= config_.population_size)
867 {
868 break;
869 }
870 }
871 }
872
873 // Evaluate offspring
874 evaluate_batch(all_children, objectives, batch_evaluator);
875
876 // Combine parent and offspring populations
877 Population combined_pop;
878 for (auto& ind : *population_)
879 {
880 combined_pop.add(ind);
881 }
882 for (auto& ind : offspring_pop)
883 {
884 combined_pop.add(ind);
885 }
886
887 // Environmental selection
888 Population new_population;
889 environmental_selection(combined_pop, new_population);
890 population_ = std::make_shared<Population>(std::move(new_population));
891
892 // Callback
893 if (callback)
894 {
895 // Store current parameter size before callback
896 size_t param_size_before = config_.lower_bounds.size();
897
898 callback(gen, *population_);
899
900 // Check if parameter space was changed during callback
901 size_t param_size_after = config_.lower_bounds.size();
902 if (param_size_before != param_size_after)
903 {
904 std::cout << "Parameter space changed during callback: "
905 << param_size_before << " -> " << param_size_after << std::endl;
906
907 // Verify population consistency
908 bool all_consistent = true;
909 for (size_t i = 0; i < population_->size(); ++i)
910 {
911 if ((*population_)[i]->parameters.size() != param_size_after)
912 {
913 std::cerr << "ERROR: Individual " << i << " has inconsistent size after callback" << std::endl;
914 all_consistent = false;
915 }
916 // Clear objectives to force re-evaluation
917 (*population_)[i]->objectives.clear();
918 (*population_)[i]->rank = 0;
919 (*population_)[i]->crowding_distance = 0.0;
920 }
921
922 if (!all_consistent)
923 {
924 throw std::runtime_error("Population inconsistency after parameter space change");
925 }
926
927 std::cout << "Re-evaluating population with new parameter space" << std::endl;
928 std::vector<IndividualPtr> individuals_to_evaluate;
929 for (auto& ind : *population_) {
930 individuals_to_evaluate.push_back(ind);
931 }
932 evaluate_batch(individuals_to_evaluate, objectives, batch_evaluator);
933 }
934 }
935 }
936
937 // Extract final Pareto front
938 auto pareto_front = get_pareto_front();
939
940 std::vector<std::vector<double>> final_params;
941 std::vector<std::vector<double>> final_objectives;
942
943 for (const auto& ind : pareto_front)
944 {
945 final_params.push_back(ind->parameters);
946 final_objectives.push_back(ind->objectives);
947 }
948
949 return {final_params, final_objectives};
950}
